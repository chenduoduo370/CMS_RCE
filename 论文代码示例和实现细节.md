### 4.1 多资源指纹识别 - 代码示例

#### 4.1.1 指纹识别模块的核心实现

```python
# fingerprint.py - 指纹识别模块的关键部分

import hashlib
import requests
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import Dict, List, Optional

class FingerprintCollector:
    """指纹收集器 - 支持多资源类型和并发下载"""

    def __init__(self, timeout: int = 3, max_workers: int = 50):
        self.timeout = timeout
        self.max_workers = max_workers
        self.session = requests.Session()

    def calculate_hash(self, content: bytes, algorithm: str = 'md5') -> str:
        """计算内容的哈希值"""
        if algorithm == 'md5':
            return hashlib.md5(content).hexdigest()
        elif algorithm == 'sha1':
            return hashlib.sha1(content).hexdigest()
        elif algorithm == 'sha256':
            return hashlib.sha256(content).hexdigest()
        return ""

    def download_resource(self, url: str) -> Optional[bytes]:
        """下载单个资源"""
        try:
            response = self.session.get(url, timeout=self.timeout)
            if response.status_code == 200:
                return response.content
        except Exception as e:
            print(f"下载失败 {url}: {e}")
        return None

    def collect_fingerprints(self, base_url: str,
                            resource_urls: List[str]) -> Dict[str, Dict]:
        """并发收集多个资源的指纹"""
        fingerprints = {}

        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # 提交所有下载任务
            future_to_url = {
                executor.submit(self.download_resource, url): url
                for url in resource_urls
            }

            # 处理完成的任务
            for future in as_completed(future_to_url):
                url = future_to_url[future]
                try:
                    content = future.result()
                    if content:
                        fingerprints[url] = {
                            'md5': self.calculate_hash(content, 'md5'),
                            'sha1': self.calculate_hash(content, 'sha1'),
                            'sha256': self.calculate_hash(content, 'sha256'),
                            'size': len(content)
                        }
                except Exception as e:
                    print(f"处理失败 {url}: {e}")

        return fingerprints

    def get_http_headers_fingerprint(self, url: str) -> Dict[str, str]:
        """获取HTTP头指纹"""
        try:
            response = self.session.head(url, timeout=self.timeout)
            return {
                'Server': response.headers.get('Server', ''),
                'X-Powered-By': response.headers.get('X-Powered-By', ''),
                'X-Generator': response.headers.get('X-Generator', ''),
                'Content-Type': response.headers.get('Content-Type', '')
            }
        except Exception as e:
            print(f"获取HTTP头失败: {e}")
        return {}
```

**代码说明**：
- `calculate_hash()`: 支持MD5、SHA1、SHA256三种哈希算法
- `download_resource()`: 单个资源下载，包含超时控制
- `collect_fingerprints()`: 使用ThreadPoolExecutor实现并发下载，提升效率
- `get_http_headers_fingerprint()`: 获取HTTP头信息用于指纹识别

#### 4.1.2 指纹匹配算法

```python
# fingerprint_cve_mapping.py - 指纹与CVE的匹配

import json
from typing import List, Dict

class FingerprintCVEMatcher:
    """指纹-CVE匹配器"""

    def __init__(self, mapping_file: str):
        """加载指纹-CVE映射数据库"""
        with open(mapping_file, 'r') as f:
            self.mapping = json.load(f)

    def match_exact(self, fingerprint: str) -> List[str]:
        """精确匹配 - 直接查找指纹"""
        cves = []
        for entry in self.mapping:
            if entry.get('fingerprint') == fingerprint:
                cves.append(entry.get('cve_id'))
        return cves

    def match_fuzzy(self, fingerprint: str, threshold: float = 0.8) -> List[str]:
        """模糊匹配 - 使用编辑距离算法"""
        from difflib import SequenceMatcher

        cves = []
        for entry in self.mapping:
            db_fingerprint = entry.get('fingerprint', '')
            similarity = SequenceMatcher(
                None, fingerprint, db_fingerprint
            ).ratio()

            if similarity >= threshold:
                cves.append(entry.get('cve_id'))

        return cves

    def match_by_headers(self, headers: Dict[str, str]) -> List[str]:
        """基于HTTP头的匹配"""
        cves = []
        server = headers.get('Server', '').lower()

        for entry in self.mapping:
            if 'server_pattern' in entry:
                pattern = entry['server_pattern'].lower()
                if pattern in server:
                    cves.append(entry.get('cve_id'))

        return cves
```

**代码说明**：
- `match_exact()`: 精确匹配，适用于已知指纹
- `match_fuzzy()`: 模糊匹配，使用SequenceMatcher计算相似度
- `match_by_headers()`: 基于HTTP头的模式匹配

```python
# src/core/ai_skill.py - AI验证模块

import json
import logging
from typing import Dict, Optional, Any
from openai import OpenAI

logger = logging.getLogger(__name__)

# 系统提示词
VERIFICATION_SYSTEM_PROMPT = """你是一个网络安全专家，专门分析漏洞利用的HTTP响应。
你的任务是判断响应内容是否表明漏洞利用成功。

判断标准：
1. 真实漏洞回显：包含命令执行结果、系统信息、错误堆栈等
2. WAF拦截页面：包含"403 Forbidden"、"blocked"、"安全拦截"等
3. 语义假阳性：页面正常但恰好包含关键字（如文档中的"root"）

返回JSON格式：
{
  "is_vulnerable": true/false,
  "confidence": 0-100,
  "reason": "判断依据的详细说明"
}"""

class AIVerificationEngine:
    """AI验证引擎 - 使用GPT-4进行语义分析"""

    def __init__(self, api_key: str, model: str = "gpt-4"):
        self.client = OpenAI(api_key=api_key)
        self.model = model

    def verify_response(self,
                       cve_id: str,
                       payload_sent: str,
                       http_response: str,
                       max_tokens: int = 500) -> Optional[Dict[str, Any]]:
        """
        使用AI验证漏洞利用响应

        Args:
            cve_id: CVE编号
            payload_sent: 发送的Payload内容
            http_response: HTTP响应文本
            max_tokens: 最大token数

        Returns:
            验证结果字典，包含is_vulnerable、confidence、reason
        """

        # 构造用户提示词
        user_prompt = f"""
CVE编号: {cve_id}
发送的Payload: {payload_sent[:500]}
HTTP响应内容:
{http_response[:2000]}

请分析这个响应是否表明漏洞利用成功。
"""

        try:
            # 调用GPT-4 API
            response = self.client.chat.completions.create(
                model=self.model,
                messages=[
                    {"role": "system", "content": VERIFICATION_SYSTEM_PROMPT},
                    {"role": "user", "content": user_prompt}
                ],
                max_tokens=max_tokens,
                temperature=0.3,  # 降低温度以获得更一致的结果
                timeout=30
            )

            # 解析响应
            result_text = response.choices[0].message.content
            result = json.loads(result_text)

            return {
                "is_vulnerable": result.get("is_vulnerable", False),
                "confidence": result.get("confidence", 0),
                "reason": result.get("reason", ""),
                "model": self.model
            }

        except json.JSONDecodeError:
            logger.error("AI响应格式错误，无法解析JSON")
            return None
        except Exception as e:
            logger.error(f"AI验证失败: {e}")
            return None

    def batch_verify(self,
                    test_cases: List[Dict]) -> List[Dict]:
        """批量验证多个测试用例"""
        results = []
        for case in test_cases:
            result = self.verify_response(
                case['cve_id'],
                case['payload'],
                case['response']
            )
            if result:
                results.append(result)
        return results
```

**代码说明**：
- `verify_response()`: 单个响应验证，调用GPT-4 API
- 系统提示词定义了AI的角色和判断标准
- 用户提示词包含CVE ID、Payload内容、HTTP响应
- 返回JSON格式的验证结果，包含是否存在漏洞、置信度、判断依据
- `batch_verify()`: 支持批量验证

#### 4.2.2 降级处理和缓存机制

```python
# 降级处理 - 当AI验证失败时使用传统方法

class HybridVerificationEngine:
    """混合验证引擎 - 支持AI验证和传统验证的降级"""

    def __init__(self, ai_engine: AIVerificationEngine):
        self.ai_engine = ai_engine
        self.cache = {}  # 简单的内存缓存

    def verify_with_fallback(self,
                            cve_id: str,
                            payload_sent: str,
                            http_response: str) -> Dict[str, Any]:
        """
        带降级的验证 - 优先使用AI，失败时降级到传统方法
        """

        # 检查缓存
        cache_key = f"{cve_id}:{hash(http_response)}"
        if cache_key in self.cache:
            return self.cache[cache_key]

        # 尝试AI验证
        ai_result = self.ai_engine.verify_response(
            cve_id, payload_sent, http_response
        )

        if ai_result:
            self.cache[cache_key] = ai_result
            return ai_result

        # 降级到传统验证
        logger.warning("AI验证失败，降级到传统验证方法")
        traditional_result = self._traditional_verify(
            cve_id, http_response
        )

        self.cache[cache_key] = traditional_result
        return traditional_result

    def _traditional_verify(self,
                           cve_id: str,
                           http_response: str) -> Dict[str, Any]:
        """传统验证方法 - 基于正则表达式"""
        import re

        # CVE特定的验证规则
        rules = {
            'CVE-2019-6340': [
                r'uid=\d+',  # 命令执行结果
                r'gid=\d+',
                r'groups=\d+'
            ],
            'CVE-2018-7600': [
                r'uid=\d+',
                r'root',
                r'bin/bash'
            ]
        }

        confidence = 0
        matched_patterns = []

        if cve_id in rules:
            for pattern in rules[cve_id]:
                if re.search(pattern, http_response):
                    confidence += 30
                    matched_patterns.append(pattern)

        # 检查WAF拦截
        if re.search(r'403|blocked|安全拦截', http_response):
            confidence = max(0, confidence - 20)

        return {
            "is_vulnerable": confidence >= 50,
            "confidence": min(100, confidence),
            "reason": f"匹配规则: {', '.join(matched_patterns)}",
            "method": "traditional"
        }
```

**代码说明**：
- `verify_with_fallback()`: 优先使用AI验证，失败时降级
- 实现了简单的内存缓存机制
- `_traditional_verify()`: 传统的正则表达式验证方法
- 支持CVE特定的验证规则

import importlib
import requests
from typing import Dict, Optional, Any

class PayloadManager:
    """Payload管理器 - 统一的payload操作接口"""

    def __init__(self, debug: bool = False):
        self.debug = debug
        self.session = requests.Session()

    def load_payload_module(self, cve_id: str):
        """动态加载Payload模块"""
        try:
            # 将CVE ID转换为模块名
            # CVE-2019-6340 -> CVE_2019_6340
            module_name = cve_id.replace('-', '_')

            # 动态导入模块
            module = importlib.import_module(f'payloads.{module_name}')
            return module
        except ImportError as e:
            print(f"无法加载Payload模块 {cve_id}: {e}")
            return None

    def build_payload(self,
                     cve_id: str,
                     target: str,
                     command: str) -> Optional[Dict[str, Any]]:
        """构建Payload"""
        module = self.load_payload_module(cve_id)
        if not module or not hasattr(module, 'build'):
            return None

        try:
            payload = module.build(target, command)
            return payload
        except Exception as e:
            print(f"构建Payload失败: {e}")
            return None

    def send_payload(self,
                    payload: Dict[str, Any],
                    timeout: int = 10) -> Optional[requests.Response]:
        """发送Payload"""
        try:
            method = payload.get('method', 'GET').upper()
            url = payload.get('url')
            headers = payload.get('headers', {})
            data = payload.get('data')

            if method == 'GET':
                response = self.session.get(
                    url, headers=headers, timeout=timeout
                )
            elif method == 'POST':
                response = self.session.post(
                    url, headers=headers, data=data, timeout=timeout
                )
            else:
                return None

            return response
        except Exception as e:
            print(f"发送Payload失败: {e}")
            return None

    def execute_exploit(self,
                       cve_id: str,
                       target: str,
                       command: str) -> Optional[str]:
        """执行完整的漏洞利用流程"""
        # 构建Payload
        payload = self.build_payload(cve_id, target, command)
        if not payload:
            return None

        # 发送Payload
        response = self.send_payload(payload)
        if not response:
            return None

        # 返回响应内容
        return response.text
```

**代码说明**：
- `load_payload_module()`: 动态导入Payload模块
- `build_payload()`: 调用Payload模块的build函数构建请求
- `send_payload()`: 发送HTTP请求
- `execute_exploit()`: 完整的漏洞利用流程

# port_scanner.py - 端口扫描模块

import socket
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List, Dict

class PortScanner:
    """多线程端口扫描器"""

    def __init__(self, timeout: int = 3, max_workers: int = 50):
        self.timeout = timeout
        self.max_workers = max_workers

    def scan_port(self, host: str, port: int) -> bool:
        """扫描单个端口"""
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(self.timeout)
            result = sock.connect_ex((host, port))
            sock.close()
            return result == 0
        except Exception:
            return False

    def scan_ports(self, host: str, ports: List[int]) -> Dict[int, bool]:
        """并发扫描多个端口"""
        open_ports = {}

        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            future_to_port = {
                executor.submit(self.scan_port, host, port): port
                for port in ports
            }

            for future in as_completed(future_to_port):
                port = future_to_port[future]
                try:
                    is_open = future.result()
                    if is_open:
                        open_ports[port] = True
                except Exception:
                    pass

        return open_ports

    def scan_common_ports(self, host: str) -> List[int]:
        """扫描常见端口"""
        common_ports = [
            80, 443, 8080, 8443, 3306, 5432, 6379, 27017,
            22, 21, 25, 53, 110, 143, 3389, 5900
        ]

        open_ports = self.scan_ports(host, common_ports)
        return sorted(open_ports.keys())
```

**代码说明**：
- `scan_port()`: 使用socket进行TCP连接测试
- `scan_ports()`: 使用ThreadPoolExecutor实现并发扫描
- `scan_common_ports()`: 扫描常见的服务端口

from functools import lru_cache

class CachedFingerprintCollector(FingerprintCollector):
    """支持缓存的指纹收集器"""

    @lru_cache(maxsize=1000)
    def calculate_hash(self, content: bytes, algorithm: str = 'md5') -> str:
        """缓存哈希计算结果"""
        return super().calculate_hash(content, algorithm)

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.fingerprint_cache = {}

    def collect_fingerprints_cached(self, base_url: str,
                                   resource_urls: List[str]) -> Dict:
        """支持缓存的指纹收集"""
        cache_key = f"{base_url}:{hash(tuple(resource_urls))}"

        if cache_key in self.fingerprint_cache:
            return self.fingerprint_cache[cache_key]

        result = self.collect_fingerprints(base_url, resource_urls)
        self.fingerprint_cache[cache_key] = result

        return result

        total=3,
        backoff_factor=1,
        status_forcelist=[429, 500, 502, 503, 504],
        allowed_methods=["HEAD", "GET", "OPTIONS"]
    )

    # 配置HTTP和HTTPS适配器
    adapter = HTTPAdapter(max_retries=retry_strategy, pool_connections=50, pool_maxsize=50)
    session.mount("http://", adapter)
    session.mount("https://", adapter)

    return session
